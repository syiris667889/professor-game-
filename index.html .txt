        let currentIndex = 0;
        let timerInterval = null;
        let timerState = 'idle'; // idle, running, finished
        let timeLeft = 5;

        function render() {
            if (timerInterval) clearInterval(timerInterval);
            timerState = 'idle';
            timeLeft = 5;

            const slide = slides[currentIndex];
            const container = document.getElementById('content-container');
            
            document.getElementById('round-indicator').innerHTML = `<span class="w-1.5 h-1.5 rounded-full bg-emerald-500"></span><span>${slide.round}</span>`;
            document.getElementById('progress-indicator').innerText = `${currentIndex + 1} / ${slides.length}`;

            let html = '';

            if (slide.type === 'title') {
                html = `
                    <div class="space-y-6">
                        <p class="text-zinc-500 text-sm tracking-widest uppercase font-semibold">${slide.subtitle}</p>
                        <h1 class="text-5xl md:text-8xl font-bold tracking-tight text-white">${slide.title}</h1>
                        <p class="text-zinc-600 text-xs tracking-widest uppercase pt-8">Нажмите Пробел для старта</p>
                    </div>
                `;
            } else if (slide.type === 'round') {
                html = `
                    <div class="space-y-4">
                        <p class="text-zinc-500 text-sm tracking-widest uppercase font-semibold">${slide.round}</p>
                        <h2 class="text-4xl md:text-7xl font-bold tracking-tight text-white">${slide.title}</h2>
                        ${slide.note ? `<p class="text-amber-400 text-sm font-medium pt-2">⚠️️ ${slide.note}</p>` : ''}
                    </div>
                `;
            } else if (slide.type === 'task') {
                html = `
                    <div class="space-y-10">
                        <h2 class="text-3xl md:text-6xl font-semibold tracking-tight text-white leading-relaxed max-w-4xl mx-auto">${slide.text}</h2>
                        <div id="timer-display" class="py-2">
                            <span id="timer-number" class="text-7xl md:text-9xl font-bold tracking-tighter text-zinc-400 font-mono">5</span>
                            <p id="timer-hint" class="text-xs text-zinc-600 uppercase tracking-widest mt-3">Пробел — запустить таймер</p>
                        </div>
                    </div>
                `;
            } else if (slide.type === 'question') {
                html = `
                    <div class="space-y-8">
                        <h2 class="text-3xl md:text-6xl font-semibold tracking-tight text-white leading-relaxed max-w-4xl mx-auto">${slide.text}</h2>
                        ${slide.warning ? `<p class="text-red-400 text-xs tracking-widest uppercase font-medium">⚠️ ${slide.warning}</p>` : ''}
                    </div>
                `;
            } else if (slide.type === 'final') {
                html = `
                    <div class="space-y-6">
                        <p class="text-zinc-500 text-sm tracking-widest uppercase font-semibold">${slide.round}</p>
                        <h2 class="text-3xl md:text-5xl font-bold tracking-tight text-white">${slide.title}</h2>
                        <p class="text-2xl md:text-4xl font-normal text-zinc-300 italic pt-4 max-w-3xl mx-auto">${slide.text}</p>
                    </div>
                `;
            }

            container.innerHTML = html;
        }

        function triggerAction() {
            const slide = slides[currentIndex];
            if (slide.type === 'task') {
                if (timerState === 'idle') {
                    startTimer();
                } else if (timerState === 'finished') {
                    nextSlide();
                }
            } else {
                nextSlide();
            }
        }

        function startTimer() {
            timerState = 'running';
            timeLeft = 5;
            const numEl = document.getElementById('timer-number');
            const hintEl = document.getElementById('timer-hint');

            numEl.innerText = timeLeft;
            numEl.classList.remove('text-zinc-400');
            numEl.classList.add('text-white', 'scale-110');
            hintEl.innerText = 'Время идет...';

            timerInterval = setInterval(() => {
                timeLeft--;
                if (timeLeft > 0) {
                    numEl.innerText = timeLeft;
                } else {
                    numEl.innerText = 'Время!';
                    numEl.classList.remove('text-white');
                    numEl.classList.add('text-red-500');
                    hintEl.innerText = 'Пробел — далее';
                    timerState = 'finished';
                    clearInterval(timerInterval);
                }
            }, 1000);
        }

        function nextSlide() {
            if (currentIndex < slides.length - 1) {
                currentIndex++;
                render();
            }
        }

        function prevSlide() {
            if (currentIndex > 0) {
                currentIndex--;
                render();
            }
        }

        function toggleMenu() {
            const modal = document.getElementById('menu-modal');
            modal.classList.toggle('hidden');
            if (!modal.classList.contains('hidden')) {
                buildMenu();
            }
        }

        function buildMenu() {
            const list = document.getElementById('slides-list');
            list.innerHTML = slides.map((s, i) => `
                <div onclick="goToSlide(${i})" class="p-2 hover:bg-zinc-800 rounded cursor-pointer flex justify-between items-center ${i === currentIndex ? 'text-white font-bold bg-zinc-800/50' : 'text-zinc-400'}">
                    <span>${i + 1}. ${s.title || s.text || s.round}</span>
                    <span class="text-xs text-zinc-600">${s.round}</span>
                </div>
            `).join('');
        }

        function goToSlide(i) {
            currentIndex = i;
            toggleMenu();
            render();
        }

        window.addEventListener('keydown', (e) => {
            if (e.code === 'Space') {
                e.preventDefault();
                triggerAction();
            } else if (e.code === 'ArrowRight') {
                nextSlide();
            } else if (e.code === 'ArrowLeft') {
                prevSlide();
            }
        });

        window.onload = () => {
            render();
        };
    </script>
